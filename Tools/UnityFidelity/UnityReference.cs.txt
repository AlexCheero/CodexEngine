// post-processing and material data. capture_unity.ps1 installs it temporarily into the source project.

    // probeAt: x,y,z triples whose interpolated light probe and probe occlusion go to .lights.txt, to check what a
    // moving object receives at a given spot.
    [Serializable] public class SceneEntry { public string name, scene; public string[] additive = new string[0]; public string cameraPrefab; public bool forceCameraPrefab; public float[] probeAt = new float[0]; }
    [Serializable] public class Config { public SceneEntry[] scenes = new SceneEntry[0]; public int quality = -1; }
    static string Env(string key) => Environment.GetEnvironmentVariable("CODEX_FIDELITY_" + key);
    static string experiment = "";

    public static void Run()
    {
        try{
            string native=Env("POSES"),output=Env("OUT");
            var config=JsonUtility.FromJson<Config>(File.ReadAllText(Env("CONFIG")));
            Directory.CreateDirectory(output);
            ShaderUtil.allowAsyncCompilation=false;
            int quality=int.Parse(Env("QUALITY")??config.quality.ToString());
            if(quality>=0)QualitySettings.SetQualityLevel(quality,true);
            foreach(var entry in config.scenes){string level=entry.name;
                string pose=Path.Combine(native,level+".png.camera.json");
                if(!File.Exists(pose)){Debug.Log("FIDELITY skip "+level);continue;}
                // CODEX_FIDELITY_EXPERIMENT may list several experiments; one editor session renders them all.
                foreach(var name in (Env("EXPERIMENT")??"").Split(',')){
                    experiment=name.Trim();
                    Capture(entry,File.ReadAllText(pose),Path.Combine(output,level+(experiment.Length==0?"":"."+experiment)+".png"));
                }
            }
            Debug.Log("FIDELITY REFERENCE COMPLETE");
            EditorApplication.Exit(0);
        }catch(Exception e){Debug.LogException(e);EditorApplication.Exit(1);}
    }
    static Vector3 Vec(string json,string key)
    {
        var m=Regex.Match(json,"\""+key+"\":\\[([^\\]]+)\\]");
        var p=m.Groups[1].Value.Split(',').Select(v=>float.Parse(v,CultureInfo.InvariantCulture)).ToArray();
        return new Vector3(p[0],p[1],p[2]);
    }
    static float Num(string json,string key)=>float.Parse(Regex.Match(json,"\""+key+"\":([-0-9.eE]+)").Groups[1].Value,CultureInfo.InvariantCulture);
    static string F(float v)=>v.ToString("R",CultureInfo.InvariantCulture);
    static string V4(Vector4 v)=>$"[{F(v.x)},{F(v.y)},{F(v.z)},{F(v.w)}]";
    static string C(Color c)=>$"[{F(c.r)},{F(c.g)},{F(c.b)},{F(c.a)}]";
    // Writes the blended URP volume stack seen by the camera (default profile plus scene volumes).
    static void DumpStack(Camera cam,UniversalAdditionalCameraData data,string path)
    {
        var s=VolumeManager.instance.CreateStack();
        VolumeManager.instance.Update(s,cam.transform,data.volumeLayerMask);
        var b=s.GetComponent<Bloom>();var t=s.GetComponent<Tonemapping>();var ca=s.GetComponent<ColorAdjustments>();
        var lgg=s.GetComponent<LiftGammaGain>();var smh=s.GetComponent<ShadowsMidtonesHighlights>();var v=s.GetComponent<Vignette>();
        var ch=s.GetComponent<ChromaticAberration>();var wb=s.GetComponent<WhiteBalance>();var fg=s.GetComponent<FilmGrain>();
        var asset=(UniversalRenderPipelineAsset)GraphicsSettings.currentRenderPipeline;
        string json="{"+
            $"\"renderScale\":{F(asset.renderScale)},\"msaa\":{asset.msaaSampleCount},\"hdr\":{(cam.allowHDR&&asset.supportsHDR?1:0)},\"postProcessing\":{(data.renderPostProcessing?1:0)},"+
            $"\"tonemap\":{(int)t.mode.value},"+
            $"\"postExposure\":{F(ca.postExposure.value)},\"contrast\":{F(ca.contrast.value)},\"saturation\":{F(ca.saturation.value)},"+
            $"\"hueShift\":{F(ca.hueShift.value)},\"colorFilter\":{C(ca.colorFilter.value)},"+
            $"\"temperature\":{F(wb.temperature.value)},\"tint\":{F(wb.tint.value)},"+
            $"\"lift\":{V4(lgg.lift.value)},\"gamma\":{V4(lgg.gamma.value)},\"gain\":{V4(lgg.gain.value)},"+
            $"\"shadows\":{V4(smh.shadows.value)},\"midtones\":{V4(smh.midtones.value)},\"highlights\":{V4(smh.highlights.value)},"+
            $"\"smhLimits\":[{F(smh.shadowsStart.value)},{F(smh.shadowsEnd.value)},{F(smh.highlightsStart.value)},{F(smh.highlightsEnd.value)}],"+
            $"\"bloom\":[{F(b.intensity.value)},{F(b.threshold.value)},{F(b.scatter.value)},{F(b.clamp.value)}],\"bloomTint\":{C(b.tint.value)},\"bloomHQ\":{(b.highQualityFiltering.value?1:0)},\"bloomMax\":{b.maxIterations.value},\"bloomDownscale\":{(int)b.downscale.value},"+
            $"\"vignette\":[{F(v.intensity.value)},{F(v.smoothness.value)},{(v.rounded.value?1:0)},{F(v.center.value.x)},{F(v.center.value.y)}],\"vignetteColor\":{C(v.color.value)},"+
            $"\"chromatic\":{F(ch.intensity.value)},\"grain\":{F(fg.intensity.value)}"+
            "}";
        File.WriteAllText(path,json);
    }
    // Lists the renderer and material under a grid of screen points.
    static void Probe(Camera cam,string path)
    {
        Physics.SyncTransforms();
        var lines=new System.Collections.Generic.List<string>();
        // Default 3x4 grid, or CODEX_FIDELITY_PROBE="x,y;x,y" viewport points measured from the top-left.
        var points=(Env("PROBE")??"").Split(';',StringSplitOptions.RemoveEmptyEntries).Select(p=>p.Split(',')).Select(p=>(float.Parse(p[0],CultureInfo.InvariantCulture),float.Parse(p[1],CultureInfo.InvariantCulture))).ToList();
        if(points.Count==0)foreach(var gy in new[]{.15f,.4f,.6f,.85f})foreach(var gx in new[]{.15f,.5f,.85f})points.Add((gx,gy));
        foreach(var (vx,vy) in points){
            var ray=cam.ViewportPointToRay(new Vector3(vx,1-vy,0));
            foreach(var other in Physics.RaycastAll(ray,2000f,~0,QueryTriggerInteraction.Ignore).OrderBy(h=>h.distance).Skip(1).Take(3))lines.Add($"   behind: {other.collider.name} d={other.distance:F1}");
            var mesh=PickRenderer(ray,out float meshDistance);
            if(mesh)lines.Add($"({vx},{vy}) renderer {mesh.name} d={meshDistance:F1} key={MeshKey(mesh.GetComponent<MeshFilter>().sharedMesh,mesh.sharedMaterials)} mats={string.Join(",",mesh.sharedMaterials.Select(m=>m?m.name+"/"+m.shader.name:"null"))} static={mesh.gameObject.isStatic} shadows={mesh.shadowCastingMode} layer={LayerMask.LayerToName(mesh.gameObject.layer)}");
            if(!Physics.Raycast(ray,out var hit,2000f,~0,QueryTriggerInteraction.Ignore)){lines.Add($"({vx},{vy}) no collider");continue;}
            var r=hit.collider.GetComponentInParent<Renderer>()??hit.collider.GetComponentInChildren<Renderer>();
            string info=$"({vx},{vy}) {hit.collider.name} d={hit.distance:F1}";
            if(r!=null){
                info+=$" renderer={r.name} lm={r.lightmapIndex} probes={r.lightProbeUsage}";
                foreach(var m in r.sharedMaterials){
                    if(m==null)continue;info+=$" || mat={m.name} shader={m.shader.name}";
                    foreach(var prop in new[]{"_BaseColor","_Color","_EmissionColor","_SpecColor"})if(m.HasProperty(prop))info+=$" {prop}={m.GetColor(prop)}";
                    foreach(var prop in new[]{"_Smoothness","_Glossiness","_Metallic","_BumpScale","_Cutoff","_Surface","_WorkflowMode","_SpecularHighlights","_EnvironmentReflections"})if(m.HasProperty(prop))info+=$" {prop}={m.GetFloat(prop)}";
                    info+=" kw="+string.Join(",",m.shaderKeywords);
                }
            }
            lines.Add(info);
        }
        File.WriteAllLines(path,lines);
    }
    // Per native mesh key (same hash as WorldExport.Model): URP Lit surface parameters per submesh.
    static string MeshKey(Mesh mesh,Material[] mats){
        string Id(UnityEngine.Object o){AssetDatabase.TryGetGUIDAndLocalFileIdentifier(o,out string guid,out long id);return guid+"~"+id+"~"+o.name;}
        var key=Id(mesh)+"|"+string.Join("|",mats.Select(x=>x?Id(x):""));
        using var sha=System.Security.Cryptography.SHA256.Create();
        return "Meshes/"+BitConverter.ToString(sha.ComputeHash(System.Text.Encoding.UTF8.GetBytes(key)),0,10).Replace("-","").ToLowerInvariant()+".brm";
    }
    static readonly System.Collections.Generic.Dictionary<Texture,Color> averages=new System.Collections.Generic.Dictionary<Texture,Color>();
    static Color Average(Texture t){
        if(averages.TryGetValue(t,out var c))return c;
        var rt=RenderTexture.GetTemporary(64,64,0,RenderTextureFormat.ARGBFloat,RenderTextureReadWrite.Linear);
        Graphics.Blit(t,rt);RenderTexture.active=rt;
        var copy=new Texture2D(64,64,TextureFormat.RGBAFloat,false,true);copy.ReadPixels(new Rect(0,0,64,64),0,0);RenderTexture.active=null;RenderTexture.ReleaseTemporary(rt);
        var px=copy.GetPixels();UnityEngine.Object.DestroyImmediate(copy);
        c=new Color(px.Average(x=>x.r),px.Average(x=>x.g),px.Average(x=>x.b),px.Average(x=>x.a));averages[t]=c;return c;
    }
    // The albedo texture: URP names first, then a Shader Graph texture property whose display name reads as a base map.
    static readonly string[] baseKeys={"_BaseMap","_MainTex","_BaseColorMap","_Albedo","_AlbedoMap","_Diffuse","_BaseTexture"};
    public static Texture BaseMap(Material m){
        foreach(var k in baseKeys)if(m.HasProperty(k)&&m.GetTexture(k))return m.GetTexture(k);
        var shader=m.shader;
        for(int i=0;i<shader.GetPropertyCount();i++){
            if(shader.GetPropertyType(i)!=UnityEngine.Rendering.ShaderPropertyType.Texture)continue;
            var label=shader.GetPropertyDescription(i).ToLowerInvariant().Replace(" ","");
            if(label is "basemap" or "basecolor" or "albedo" or "maintex" or "maintexture" or "diffuse" or "colormap"){var t=m.GetTexture(shader.GetPropertyName(i));if(t)return t;}
        }
        return null;
    }
    static string SurfaceJson(Material m){
        if(!m)return "null";
        float F2(string k,float d)=>m.HasProperty(k)?m.GetFloat(k):d;
        bool lit=m.shader.name.Contains("Universal Render Pipeline/Lit")||m.shader.name.Contains("Terrain/Lit")||m.shader.name.Contains("Complex Lit");
        float smooth=F2("_Smoothness",F2("_Glossiness",.5f)),metal=F2("_Metallic",0);
        var gloss=m.HasProperty("_MetallicGlossMap")?m.GetTexture("_MetallicGlossMap"):null;
        var spec=m.HasProperty("_SpecGlossMap")?m.GetTexture("_SpecGlossMap"):null;
        bool specular=F2("_WorkflowMode",1)==0;
        bool albedoAlpha=F2("_SmoothnessTextureChannel",0)==1;
        if(albedoAlpha && m.HasProperty("_BaseMap") && m.GetTexture("_BaseMap"))smooth*=Average(m.GetTexture("_BaseMap")).a;
        else if(!specular && gloss){var a=Average(gloss);smooth*=a.a;metal=a.r;}
        else if(specular && spec){var a=Average(spec);smooth*=a.a;}
        var sc=m.HasProperty("_SpecColor")?m.GetColor("_SpecColor"):Color.black;
        if(specular && spec){var a=Average(spec);sc=new Color(a.r,a.g,a.b);}
        string Fl(float v)=>v.ToString("0.####",CultureInfo.InvariantCulture);
        var baseMap=BaseMap(m);
        return "{\"material\":\""+m.name+"\",\"baseMap\":\""+(baseMap?AssetDatabase.GetAssetPath(baseMap):"")+"\",\"cull\":"+Fl(F2("_Cull",2))+",\"shader\":\""+m.shader.name+"\",\"lit\":"+(lit?1:0)+",\"smoothness\":"+Fl(smooth)+",\"metallic\":"+Fl(metal)+",\"specular\":"+(specular?"["+Fl(sc.linear.r)+","+Fl(sc.linear.g)+","+Fl(sc.linear.b)+"]":"null")
            +",\"env\":"+(F2("_EnvironmentReflections",1)>0&&!m.IsKeywordEnabled("_ENVIRONMENTREFLECTIONS_OFF")?1:0)+",\"highlights\":"+(F2("_SpecularHighlights",1)>0&&!m.IsKeywordEnabled("_SPECULARHIGHLIGHTS_OFF")?1:0)
            // Distance-faded foliage (Terrain sample Shader Graphs): fade start/end and the colour reached at the end.
            +(m.HasProperty("Distance_Fade_Start")?",\"fade\":["+Fl(F2("Distance_Fade_Start",0))+","+Fl(F2("Distance_Fade_End",0))+"],\"fadeColor\":["+Fl(m.GetColor("Fade_Color").linear.r)+","+Fl(m.GetColor("Fade_Color").linear.g)+","+Fl(m.GetColor("Fade_Color").linear.b)+"]":"")
            +(m.HasProperty("AO_Scale")?",\"aoScale\":"+Fl(F2("AO_Scale",1)):"")+"}";
    }
    static void Surfaces(string path){
        var seen=new System.Collections.Generic.HashSet<string>();var rows=new System.Collections.Generic.List<string>();
        foreach(var r in UnityEngine.Object.FindObjectsByType<MeshRenderer>(FindObjectsSortMode.None)){
            var f=r.GetComponent<MeshFilter>();if(!f||!f.sharedMesh||!r.enabled)continue;
            var key=MeshKey(f.sharedMesh,r.sharedMaterials);if(!seen.Add(key))continue;
            rows.Add("\""+key+"\":["+string.Join(",",Enumerable.Range(0,f.sharedMesh.subMeshCount).Select(i=>SurfaceJson(r.sharedMaterials.Length>0?r.sharedMaterials[Math.Min(i,r.sharedMaterials.Length-1)]:null)))+"]");
        }
        foreach(var t in UnityEngine.Object.FindObjectsByType<Terrain>(FindObjectsSortMode.None))
            rows.Add("\"terrain:"+t.name+"\":["+string.Join(",",t.terrainData.terrainLayers.Select(l=>l?"{\"smoothness\":"+l.smoothness.ToString(CultureInfo.InvariantCulture)+",\"metallic\":"+l.metallic.ToString(CultureInfo.InvariantCulture)+",\"maskMap\":"+(l.maskMapTexture?1:0)+",\"diffuseAlphaSmooth\":"+(l.diffuseTexture&&GraphicsFormatUtility.HasAlphaChannel(l.diffuseTexture.graphicsFormat)?1:0)+"}":"null"))+"]");
        File.WriteAllText(path,"{"+string.Join(",\n",rows)+"}");
    }
    // Nearest MeshRenderer hit by a ray, by triangle test (renderers need no collider).
    static MeshRenderer PickRenderer(Ray ray,out float best){
        best=float.MaxValue;MeshRenderer pick=null;
        foreach(var r in UnityEngine.Object.FindObjectsByType<MeshRenderer>(FindObjectsSortMode.None)){
            if(!r.enabled||!r.bounds.IntersectRay(ray))continue;
            var f=r.GetComponent<MeshFilter>();if(!f||!f.sharedMesh)continue;
            var v=f.sharedMesh.vertices;var t=f.sharedMesh.triangles;var m=r.localToWorldMatrix;
            for(int i=0;i<t.Length;i+=3){
                Vector3 a=m.MultiplyPoint3x4(v[t[i]]),b=m.MultiplyPoint3x4(v[t[i+1]]),c=m.MultiplyPoint3x4(v[t[i+2]]);
                Vector3 e1=b-a,e2=c-a,pv=Vector3.Cross(ray.direction,e2);float det=Vector3.Dot(e1,pv);if(Mathf.Abs(det)<1e-8f)continue;
                float inv=1/det;Vector3 tv=ray.origin-a;float u=Vector3.Dot(tv,pv)*inv;if(u<0||u>1)continue;
                Vector3 qv=Vector3.Cross(tv,e1);float w=Vector3.Dot(ray.direction,qv)*inv;if(w<0||u+w>1)continue;
                float d=Vector3.Dot(e2,qv)*inv;if(d>0&&d<best){best=d;pick=r;}
            }
        }
        return pick;
    }
    static void Capture(SceneEntry entry,string pose,string path)
    {
        string level=entry.name;
        EditorSceneManager.OpenScene(entry.scene,OpenSceneMode.Single);
        foreach(var extra in entry.additive)EditorSceneManager.OpenScene(extra,OpenSceneMode.Additive);
        var cameras=UnityEngine.Object.FindObjectsByType<Camera>(FindObjectsInactive.Exclude,FindObjectsSortMode.None);
        Camera cam=cameras.Where(c=>c.enabled && c.targetTexture==null).OrderByDescending(c=>c.CompareTag("MainCamera")).ThenByDescending(c=>c.depth).FirstOrDefault();
        bool hasVolume=UnityEngine.Object.FindObjectsByType<Volume>(FindObjectsSortMode.None).Any(v=>v.isGlobal && v.enabled);
        if(!string.IsNullOrEmpty(entry.cameraPrefab) && (cam==null || entry.forceCameraPrefab || !hasVolume)){
            var go=(GameObject)PrefabUtility.InstantiatePrefab(AssetDatabase.LoadAssetAtPath<GameObject>(entry.cameraPrefab));
            foreach(var other in cameras)other.enabled=false;
            cam=go.GetComponentsInChildren<Camera>(true).First(c=>c.targetTexture==null);
            cam.gameObject.SetActive(true);cam.enabled=true;
        }
        foreach(var c in cameras)if(c!=cam && c.targetTexture==null)c.enabled=false;
        int w=(int)Num(pose,"width"),h=(int)Num(pose,"height");
        cam.transform.SetPositionAndRotation(Vec(pose,"eye"),Quaternion.LookRotation(Vec(pose,"forward"),Vec(pose,"up")));
        cam.fieldOfView=Num(pose,"fov");cam.aspect=(float)w/h;
        var data=cam.GetUniversalAdditionalCameraData();
        Debug.Log($"FIDELITY {level}: camera {cam.name} post={data.renderPostProcessing} aa={data.antialiasing} hdr={cam.allowHDR} volumes={UnityEngine.Object.FindObjectsByType<Volume>(FindObjectsSortMode.None).Length}");
        foreach(var ps in UnityEngine.Object.FindObjectsByType<ParticleSystem>(FindObjectsSortMode.None))
            if(ps.main.playOnAwake && ps.transform.parent?.GetComponentInParent<ParticleSystem>()==null){ps.useAutoRandomSeed=false;ps.Simulate(3f,true,true,true);}
        switch(experiment){
            case "nofog":RenderSettings.fog=false;break;
            case "skyonly":foreach(var r in UnityEngine.Object.FindObjectsByType<Renderer>(FindObjectsSortMode.None))r.enabled=false;
                foreach(var t in UnityEngine.Object.FindObjectsByType<Terrain>(FindObjectsSortMode.None))t.drawHeightmap=t.drawTreesAndFoliage=false;break;
            case "nopost":data.renderPostProcessing=false;break;
            case var keepExp when keepExp!=null && keepExp.StartsWith("postkeep-"):
            case "postidentity":case "postbloomonly":{
                string keep=experiment.StartsWith("postkeep-")?experiment.Substring(9):"";
                // A global override volume that neutralises every grading step (and bloom unless kept).
                var profile=ScriptableObject.CreateInstance<VolumeProfile>();
                if(keep!="tonemap"){var t=profile.Add<Tonemapping>(true);t.mode.value=TonemappingMode.None;}
                var ca=profile.Add<ColorAdjustments>(false);ca.hueShift.Override(0);ca.colorFilter.Override(Color.white);
                if(keep!="exposure")ca.postExposure.Override(0);if(keep!="contrast")ca.contrast.Override(0);if(keep!="saturation")ca.saturation.Override(0);
                if(keep!="lgg"){var lgg=profile.Add<LiftGammaGain>(true);lgg.lift.value=new Vector4(1,1,1,0);lgg.gamma.value=new Vector4(1,1,1,0);lgg.gain.value=new Vector4(1,1,1,0);}
                if(keep!="smh"){var smh=profile.Add<ShadowsMidtonesHighlights>(true);smh.shadows.value=smh.midtones.value=smh.highlights.value=new Vector4(1,1,1,0);}
                if(keep!="vignette"){var v=profile.Add<Vignette>(true);v.intensity.value=0;}var chroma=profile.Add<ChromaticAberration>(true);chroma.intensity.value=0;
                var wb=profile.Add<WhiteBalance>(true);wb.temperature.value=0;wb.tint.value=0;
                if(experiment!="postbloomonly"){var bl=profile.Add<Bloom>(true);bl.intensity.value=0;}
                var go=new GameObject("FidelityOverride");var vol=go.AddComponent<Volume>();vol.isGlobal=true;vol.priority=100000;vol.sharedProfile=profile;
                break;}
            case "particlesblack":case "particleswhite":{
                data.renderPostProcessing=false;
                foreach(var r in UnityEngine.Object.FindObjectsByType<Renderer>(FindObjectsSortMode.None))if(!(r is ParticleSystemRenderer))r.enabled=false;
                foreach(var t in UnityEngine.Object.FindObjectsByType<Terrain>(FindObjectsSortMode.None))t.drawHeightmap=t.drawTreesAndFoliage=false;
                cam.clearFlags=CameraClearFlags.SolidColor;cam.backgroundColor=experiment=="particlesblack"?Color.black:Color.white;
                break;}
            case "particlecolor":data.renderPostProcessing=false;foreach(var ps in UnityEngine.Object.FindObjectsByType<ParticleSystem>(FindObjectsSortMode.None)){var mm=ps.main;mm.startColor=new Color(1,0,0,.75f);var col=ps.colorOverLifetime;col.enabled=false;ps.Simulate(3f,true,true,true);}break;
            case "particlenoinstancing":data.renderPostProcessing=false;foreach(var pr in UnityEngine.Object.FindObjectsByType<ParticleSystemRenderer>(FindObjectsSortMode.None))pr.enableGPUInstancing=false;break;
            case "noparticles":data.renderPostProcessing=false;foreach(var pr in UnityEngine.Object.FindObjectsByType<ParticleSystemRenderer>(FindObjectsSortMode.None))pr.enabled=false;break;
            case "noreflect":data.renderPostProcessing=false;RenderSettings.reflectionIntensity=0;RenderSettings.customReflectionTexture=null;RenderSettings.defaultReflectionMode=DefaultReflectionMode.Custom;
                foreach(var probe in UnityEngine.Object.FindObjectsByType<ReflectionProbe>(FindObjectsSortMode.None))probe.enabled=false;break;
        }
        Debug.Log($"FIDELITY {level}: fog={RenderSettings.fog} mode={RenderSettings.fogMode} density={RenderSettings.fogDensity} start={RenderSettings.fogStartDistance} end={RenderSettings.fogEndDistance} color={RenderSettings.fogColor} skybox={RenderSettings.skybox?.shader.name} ambient={RenderSettings.ambientMode} {RenderSettings.ambientLight} far={cam.farClipPlane}");
        // "hdr": the camera's float colour without post-processing, written at -3 EV so values up to 8 stay visible.
        bool hdr=experiment=="hdr";
        if(hdr)data.renderPostProcessing=false;
        var rt=hdr?new RenderTexture(w,h,24,RenderTextureFormat.ARGBHalf,RenderTextureReadWrite.Linear){antiAliasing=1}
                  :new RenderTexture(w,h,24,RenderTextureFormat.ARGB32,RenderTextureReadWrite.sRGB){antiAliasing=1};
        var request=new RenderPipeline.StandardRequest{destination=rt};
        for(int i=0;i<4;i++){
            if(RenderPipeline.SupportsRenderRequest(cam,request))RenderPipeline.SubmitRenderRequest(cam,request);
            else{cam.targetTexture=rt;cam.Render();cam.targetTexture=null;}
        }
        RenderTexture.active=rt;
        var tex=new Texture2D(w,h,TextureFormat.RGB24,false);
        if(hdr){
            var floats=new Texture2D(w,h,TextureFormat.RGBAHalf,false,true);floats.ReadPixels(new Rect(0,0,w,h),0,0);
            tex.SetPixels(floats.GetPixels().Select(c=>new Color(c.r/8,c.g/8,c.b/8,1).gamma).ToArray());UnityEngine.Object.DestroyImmediate(floats);
        }else tex.ReadPixels(new Rect(0,0,w,h),0,0);
        tex.Apply();
        RenderTexture.active=null;
        File.WriteAllBytes(path,tex.EncodeToPNG());
        DumpStack(cam,data,Path.Combine(Path.GetDirectoryName(path),level+".post.json"));
        Probe(cam,Path.Combine(Path.GetDirectoryName(path),level+".probe.txt"));
        Surfaces(Path.Combine(Path.GetDirectoryName(path),level+".surfaces.json"));
        // LODGroups: world reference point, size and the screen-relative height below which the last LOD is culled.
        File.WriteAllLines(Path.Combine(Path.GetDirectoryName(path),level+".lods.txt"),
            UnityEngine.Object.FindObjectsByType<LODGroup>(FindObjectsSortMode.None).Where(g=>g.enabled).Select(g=>{
                var lods=g.GetLODs();var keys=lods.Length>0?lods[0].renderers.Where(r=>r is MeshRenderer&&r.GetComponent<MeshFilter>()&&r.GetComponent<MeshFilter>().sharedMesh)
                    .Select(r=>MeshKey(r.GetComponent<MeshFilter>().sharedMesh,r.sharedMaterials)+"@"+F(r.transform.position.x)+","+F(r.transform.position.y)+","+F(r.transform.position.z)):Enumerable.Empty<string>();
                var center=g.transform.TransformPoint(g.localReferencePoint);
                float scale=Mathf.Max(Mathf.Abs(g.transform.lossyScale.x),Mathf.Max(Mathf.Abs(g.transform.lossyScale.y),Mathf.Abs(g.transform.lossyScale.z)));
                return $"{g.name}|{F(center.x)},{F(center.y)},{F(center.z)}|{F(g.size*scale)}|{string.Join(",",lods.Select(l=>F(l.screenRelativeTransitionHeight)))}|{string.Join(";",keys)}";
            }).Append("lodBias="+F(QualitySettings.lodBias)));
        File.WriteAllLines(Path.Combine(Path.GetDirectoryName(path),level+".particles.txt"),
            UnityEngine.Object.FindObjectsByType<ParticleSystem>(FindObjectsSortMode.None).Select(ps=>{
                var r=ps.GetComponent<ParticleSystemRenderer>();var m=ps.main;
                return $"{ps.name} count={ps.particleCount} bounds={(r?r.bounds.center:Vector3.zero)}/{(r?r.bounds.size:Vector3.zero)} scaling={m.scalingMode} space={m.simulationSpace} pos={ps.transform.position} lossy={ps.transform.lossyScale} local={ps.transform.localScale} startSize={m.startSize.constantMin}-{m.startSize.constantMax} rate={ps.emission.rateOverTime.constantMax} life={m.startLifetime.constantMax}";
            }));
        var lm=new System.Collections.Generic.List<string>();
        for(int i=0;i<LightmapSettings.lightmaps.Length;i++){
            var t=LightmapSettings.lightmaps[i].lightmapColor;if(!t)continue;
            Color[] px;
            if(t.isReadable)px=t.GetPixels();
            else{
                var lmrt=RenderTexture.GetTemporary(t.width,t.height,0,RenderTextureFormat.ARGBFloat,RenderTextureReadWrite.Linear);
                Graphics.Blit(t,lmrt);RenderTexture.active=lmrt;
                var copy=new Texture2D(t.width,t.height,TextureFormat.RGBAFloat,false,true);copy.ReadPixels(new Rect(0,0,t.width,t.height),0,0);
                RenderTexture.active=null;RenderTexture.ReleaseTemporary(lmrt);px=copy.GetPixels();
            }
            string stats=px==null?"unreadable":$"mean={px.Average(c=>c.r):F4},{px.Average(c=>c.g):F4},{px.Average(c=>c.b):F4} a={px.Average(c=>c.a):F4} max={px.Max(c=>Mathf.Max(c.r,Mathf.Max(c.g,c.b))):F3}";
            lm.Add($"{i} {t.name} {t.format} {t.width}x{t.height} {stats} encoding={LightmapSettings.lightmapsMode} hdr={GraphicsFormatUtility.IsHDRFormat(t.graphicsFormat)}");
        }
        File.WriteAllLines(Path.Combine(Path.GetDirectoryName(path),level+".lightmaps.txt"),lm);
        var lights=new System.Collections.Generic.List<string>();
        foreach(var l in UnityEngine.Object.FindObjectsByType<Light>(FindObjectsSortMode.None)){
            if(!l.isActiveAndEnabled)continue;
            var ld=l.GetComponent<UniversalAdditionalLightData>();
            lights.Add($"{l.type} {l.name} color={l.color} intensity={l.intensity} range={l.range} bake={l.lightmapBakeType} shadows={l.shadows} strength={l.shadowStrength} bias={l.shadowBias} normalBias={l.shadowNormalBias} cull={l.cullingMask} render={l.renderMode} pos={l.transform.position} fwd={l.transform.forward}");
        }
        foreach(var t in UnityEngine.Object.FindObjectsByType<Terrain>(FindObjectsSortMode.None))
            lights.Add($"Terrain {t.name} mat={(t.materialTemplate?t.materialTemplate.shader.name:"default")} lm={t.lightmapIndex} trees={t.terrainData.treeInstanceCount} draw={t.drawTreesAndFoliage} treeDist={t.treeDistance} billboard={t.treeBillboardDistance} detailDist={t.detailObjectDistance} layers={string.Join(",",t.terrainData.terrainLayers.Select(x=>x?x.name+"/"+x.smoothness+"/"+x.metallic+"/"+x.specular:"null"))}");
        lights.Add($"ambientMode={RenderSettings.ambientMode} sky={RenderSettings.ambientSkyColor} equator={RenderSettings.ambientEquatorColor} ground={RenderSettings.ambientGroundColor} intensity={RenderSettings.ambientIntensity} reflection={RenderSettings.defaultReflectionMode} reflIntensity={RenderSettings.reflectionIntensity} bounces={RenderSettings.reflectionBounces} sun={(RenderSettings.sun?RenderSettings.sun.name:"none")} subtractive={RenderSettings.subtractiveShadowColor}");
        lights.Add($"shadowDistance={QualitySettings.shadowDistance} urpShadowDistance={((UniversalRenderPipelineAsset)GraphicsSettings.currentRenderPipeline).shadowDistance} cascades={((UniversalRenderPipelineAsset)GraphicsSettings.currentRenderPipeline).shadowCascadeCount}");
        // at: where the probe captures from. box: the world-space volume whose renderers reflect it.
        foreach(var probe in UnityEngine.Object.FindObjectsByType<ReflectionProbe>(FindObjectsSortMode.None)){
            string Exact(Vector3 v)=>"("+F(v.x)+","+F(v.y)+","+F(v.z)+")";
            lights.Add($"ReflectionProbe {probe.name} mode={probe.mode} tex={(probe.texture?probe.texture.name:"none")} intensity={probe.intensity} pos={probe.transform.position} size={probe.size} enabled={probe.isActiveAndEnabled} at={Exact(probe.transform.position)} boxCenter={Exact(probe.bounds.center)} boxSize={Exact(probe.bounds.size)} near={F(probe.nearClipPlane)} far={F(probe.farClipPlane)} boxProjection={probe.boxProjection} resolution={probe.resolution} refresh={probe.refreshMode} importance={probe.importance}");
        }
        {
            var sh=RenderSettings.ambientProbe;var dirs=new[]{Vector3.up,Vector3.down,Vector3.forward,Vector3.right};var res=new Color[4];
            sh.Evaluate(dirs,res);
            lights.Add("ambientProbe up="+res[0]+" down="+res[1]+" fwd="+res[2]+" right="+res[3]);
            var axes=new[]{Vector3.right,Vector3.left,Vector3.up,Vector3.down,Vector3.forward,Vector3.back};var cube=new Color[6];sh.Evaluate(axes,cube);
            File.WriteAllText(Path.Combine(Path.GetDirectoryName(path),level+".ambient.json"),"["+string.Join(",",cube.Select(c=>"["+c.r.ToString("R",CultureInfo.InvariantCulture)+","+c.g.ToString("R",CultureInfo.InvariantCulture)+","+c.b.ToString("R",CultureInfo.InvariantCulture)+"]"))+"]");
        }
        lights.Add($"lightProbes={(LightmapSettings.lightProbes?LightmapSettings.lightProbes.count:0)}");
        // <level>.lightprobes: the baked light probes, for CodexEngine.Rendering.IrradianceProbes.Read. After the tag and the
        // count, each probe is its position and its irradiance on the six axes (+X,-X,+Y,-Y,+Z,-Z): 3 + 18 floats.
        // .lights.txt also gets the probes' occlusion (how much of each mixed light reaches them past the static
        // casters, by shadowmask channel) as a summary, and the interpolated probe at each of the scene's probeAt points.
        if(LightmapSettings.lightProbes && LightmapSettings.lightProbes.count>0){
            var positions=LightmapSettings.lightProbes.positions;var baked=LightmapSettings.lightProbes.bakedProbes;
            var axes=new[]{Vector3.right,Vector3.left,Vector3.up,Vector3.down,Vector3.forward,Vector3.back};var cube=new Color[6];
            var interpolated=new System.Collections.Generic.List<SphericalHarmonicsL2>();var occlusion=new System.Collections.Generic.List<Vector4>();
            LightProbes.CalculateInterpolatedLightAndOcclusionProbes(new System.Collections.Generic.List<Vector3>(positions),interpolated,occlusion);
            using(var probeFile=new BinaryWriter(File.Create(Path.Combine(Path.GetDirectoryName(path),level+".lightprobes")))){
                probeFile.Write(0x42504C43);probeFile.Write(positions.Length);
                for(int probe=0;probe<positions.Length;probe++){
                    probeFile.Write(positions[probe].x);probeFile.Write(positions[probe].y);probeFile.Write(positions[probe].z);
                    baked[probe].Evaluate(axes,cube);
                    foreach(var lobe in cube){probeFile.Write(lobe.r);probeFile.Write(lobe.g);probeFile.Write(lobe.b);}
                }
            }
            lights.Add($"lightProbeOcclusion mean=({F(occlusion.Average(o=>o.x))},{F(occlusion.Average(o=>o.y))},{F(occlusion.Average(o=>o.z))},{F(occlusion.Average(o=>o.w))}) lit={occlusion.Count(o=>o.x>.5f)} of {occlusion.Count}");
            if(entry.probeAt.Length>=3){
                var at=new System.Collections.Generic.List<Vector3>();
                for(int i=0;i+2<entry.probeAt.Length;i+=3)at.Add(new Vector3(entry.probeAt[i],entry.probeAt[i+1],entry.probeAt[i+2]));
                var light=new System.Collections.Generic.List<SphericalHarmonicsL2>();var shade=new System.Collections.Generic.List<Vector4>();
                LightProbes.CalculateInterpolatedLightAndOcclusionProbes(at,light,shade);
                for(int i=0;i<at.Count;i++){
                    light[i].Evaluate(axes,cube);
                    lights.Add($"probeAt ({F(at[i].x)},{F(at[i].y)},{F(at[i].z)}) up=({F(cube[2].r)},{F(cube[2].g)},{F(cube[2].b)}) down=({F(cube[3].r)},{F(cube[3].g)},{F(cube[3].b)}) side=({F(cube[0].r)},{F(cube[0].g)},{F(cube[0].b)}) occlusion=({F(shade[i].x)},{F(shade[i].y)},{F(shade[i].z)},{F(shade[i].w)})");
                }
            }
        }
        File.WriteAllLines(Path.Combine(Path.GetDirectoryName(path),level+".lights.txt"),lights);
        Debug.Log("FIDELITY captured "+path);
    }
